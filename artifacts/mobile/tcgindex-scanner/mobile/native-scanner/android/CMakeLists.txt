cmake_minimum_required(VERSION 3.22)
project(TCGScanner)
set(CMAKE_CXX_STANDARD 20)
add_library(TCGScanner SHARED src/main/cpp/cpp-adapter.cpp src/main/cpp/CardFeatureSearch.cpp src/main/cpp/CompactFeatureSearch.cpp)
target_include_directories(TCGScanner PRIVATE src/main/cpp/third-party)
target_compile_definitions(TCGScanner PRIVATE USEARCH_USE_NUMKONG=0 USEARCH_USE_OPENMP=0)
include(../nitrogen/generated/android/TCGScanner+autolinking.cmake)
find_package(react-native-vision-camera REQUIRED CONFIG)
get_target_property(CAMERA_INCLUDE react-native-vision-camera::VisionCamera INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(TCGScanner PRIVATE ${CAMERA_INCLUDE} ${CAMERA_INCLUDE}/VisionCamera)
target_link_libraries(TCGScanner react-native-vision-camera::VisionCamera android log)
